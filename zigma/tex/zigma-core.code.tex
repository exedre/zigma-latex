%% ============================================================================
%% Link Styling (hyperref runtime policy)
%% ============================================================================

\cs_new_protected:Npn \zigma_apply_link_style:
  {
    \cs_if_exist:NT \hypersetup
      {
        % Never render boxed URL/link borders by default.
        \hypersetup{pdfborder={0 0 0}}

        \tl_set:Nx \l_tmpa_tl { \tl_lower_case:n { \tl_use:N \g_zigma_links_style_tl } }
        \tl_trim_spaces:N \l_tmpa_tl

        \str_case:VnF \l_tmpa_tl
          {
            {color}
              {
                \exp_args:Nx \hypersetup{
                  colorlinks=true,
                  urlcolor=\tl_use:N \g_zigma_links_url_color_tl,
                  linkcolor=\tl_use:N \g_zigma_links_url_color_tl,
                  citecolor=\tl_use:N \g_zigma_links_url_color_tl
                }
              }
            {mono}
              {
                \hypersetup{
                  colorlinks=true,
                  urlcolor=black,
                  linkcolor=black,
                  citecolor=black
                }
              }
            {hidden}
              {
                \hypersetup{
                  hidelinks,
                  colorlinks=false
                }
              }
          }
          {
            \msg_warning:nnx { zigma } { unknown-links-style } { \tl_use:N \g_zigma_links_style_tl }
            \exp_args:Nx \hypersetup{
              colorlinks=true,
              urlcolor=\tl_use:N \g_zigma_links_url_color_tl,
              linkcolor=\tl_use:N \g_zigma_links_url_color_tl,
              citecolor=\tl_use:N \g_zigma_links_url_color_tl
            }
          }

        \bool_if:NT \g_zigma_links_url_underline_bool
          {
            \hypersetup{
              colorlinks=false,
              pdfborder={0 0 1},
              pdfborderstyle={/S/U/W 0.7}
            }
          }
      }
  }

%% ============================================================================
%% Drop Cap Policy
%% ============================================================================
%%
%% When use.dropcap=false, neutralize \lettrine so templates/content that still
%% emit it render plain text and stay compilable across all renderers.
%%
\cs_new_protected:Npn \zigma_apply_dropcap_policy:
  {
    \bool_if:NF \g_zigma_use_dropcap_bool
      {
        \cs_if_exist:NTF \lettrine
          { \RenewDocumentCommand \lettrine { O{} m m } { ##2##3 } }
          { \NewDocumentCommand \lettrine { O{} m m } { ##2##3 } }
      }
  }

%% ============================================================================
%% License Rendering (canonical metadata output)
%% ============================================================================
%%
%% The final license text shown in renderers is always:
%%   \g_zigma_meta_license_rendered_tl
%%
%% Inputs:
%% - Legacy: metadata.license (free text)
%% - Structured: metadata.license.type/id/holder/year/url/note
%%
\cs_new_protected:Npn \zigma_meta_license_render_build:
  {
    \tl_clear:N \g_zigma_meta_license_rendered_tl
    \tl_clear:N \l_tmpa_tl
    \tl_clear:N \l_tmpb_tl
    \tl_clear:N \l_tmpc_tl
    \tl_clear:N \l_tmpd_tl

    % Canonicalized type (lowercase)
    \tl_set:Nx \l_tmpa_tl { \tl_lower_case:n { \tl_use:N \g_zigma_meta_license_type_tl } }
    \tl_trim_spaces:N \l_tmpa_tl

    % If type not provided, infer from legacy/free fields.
    \tl_if_empty:NT \l_tmpa_tl
      {
        \tl_if_empty:NF \g_zigma_meta_license_tl
          { \tl_set:Nn \l_tmpa_tl { custom } }
        \tl_if_empty:NT \l_tmpa_tl
          {
            \tl_if_empty:NF \g_zigma_meta_license_id_tl
              {
                \tl_set:Nx \l_tmpb_tl { \tl_lower_case:n { \tl_use:N \g_zigma_meta_license_id_tl } }
                \str_if_in:VnTF \l_tmpb_tl { cc }
                  { \tl_set:Nn \l_tmpa_tl { cc } }
                  {
                    \str_if_in:VnTF \l_tmpb_tl { gpl }
                      { \tl_set:Nn \l_tmpa_tl { gnu } }
                      { \tl_set:Nn \l_tmpa_tl { custom } }
                  }
              }
          }
      }

    % Type-specific canonical rendering
    \tl_if_empty:NF \l_tmpa_tl
      {
        \str_case:VnF \l_tmpa_tl
          {
            {copyright}
              {
                \tl_set:Nn \l_tmpb_tl { \the\year }
                \tl_if_empty:NF \g_zigma_meta_license_year_tl
                  { \tl_set:NV \l_tmpb_tl \g_zigma_meta_license_year_tl }

                \tl_set:Nn \l_tmpc_tl { the~author(s) }
                \tl_if_empty:NF \g_zigma_meta_license_holder_tl
                  { \tl_set:NV \l_tmpc_tl \g_zigma_meta_license_holder_tl }

                \tl_gset:Nx \g_zigma_meta_license_rendered_tl
                  { Copyright~(c)~\l_tmpb_tl~\l_tmpc_tl.~All~rights~reserved. }
              }
            {cc}
              {
                \tl_set:Nx \l_tmpb_tl { \tl_lower_case:n { \tl_use:N \g_zigma_meta_license_id_tl } }
                \tl_trim_spaces:N \l_tmpb_tl
                \tl_clear:N \l_tmpc_tl
                \tl_clear:N \l_tmpd_tl

                \str_case:VnF \l_tmpb_tl
                  {
                    {cc-by-4.0}
                      {
                        \tl_set:Nn \l_tmpc_tl { Creative~Commons~Attribution~4.0~International~(CC~BY~4.0) }
                        \tl_set:Nn \l_tmpd_tl { https://creativecommons.org/licenses/by/4.0/ }
                      }
                    {cc-by-sa-4.0}
                      {
                        \tl_set:Nn \l_tmpc_tl { Creative~Commons~Attribution-ShareAlike~4.0~International~(CC~BY-SA~4.0) }
                        \tl_set:Nn \l_tmpd_tl { https://creativecommons.org/licenses/by-sa/4.0/ }
                      }
                    {cc-by-nc-4.0}
                      {
                        \tl_set:Nn \l_tmpc_tl { Creative~Commons~Attribution-NonCommercial~4.0~International~(CC~BY-NC~4.0) }
                        \tl_set:Nn \l_tmpd_tl { https://creativecommons.org/licenses/by-nc/4.0/ }
                      }
                    {cc0-1.0}
                      {
                        \tl_set:Nn \l_tmpc_tl { Creative~Commons~CC0~1.0~Universal~(CC0~1.0) }
                        \tl_set:Nn \l_tmpd_tl { https://creativecommons.org/publicdomain/zero/1.0/ }
                      }
                  }
                  {
                    \tl_if_empty:NTF \l_tmpb_tl
                      {
                        \tl_set:Nn \l_tmpc_tl { Creative~Commons~license }
                      }
                      {
                        \msg_warning:nnxx { zigma } { unknown-license-id } { \tl_use:N \g_zigma_meta_license_id_tl } { cc }
                        \tl_set:Nx \l_tmpc_tl { Creative~Commons~license~(\tl_use:N \g_zigma_meta_license_id_tl) }
                      }
                  }

                \tl_gset:Nx \g_zigma_meta_license_rendered_tl
                  { This~work~is~licensed~under~\l_tmpc_tl. }

                \tl_if_empty:NT \g_zigma_meta_license_url_tl
                  { \tl_if_empty:NF \l_tmpd_tl { \tl_gset:NV \g_zigma_meta_license_url_tl \l_tmpd_tl } }
              }
            {gnu}
              {
                \tl_set:Nx \l_tmpb_tl { \tl_lower_case:n { \tl_use:N \g_zigma_meta_license_id_tl } }
                \tl_trim_spaces:N \l_tmpb_tl
                \tl_clear:N \l_tmpc_tl
                \tl_clear:N \l_tmpd_tl

                \str_case:VnF \l_tmpb_tl
                  {
                    {gpl-3.0-or-later}
                      {
                        \tl_set:Nn \l_tmpc_tl { GNU~General~Public~License~v3.0~or~later }
                        \tl_set:Nn \l_tmpd_tl { https://www.gnu.org/licenses/gpl-3.0.html }
                      }
                    {gpl-2.0-or-later}
                      {
                        \tl_set:Nn \l_tmpc_tl { GNU~General~Public~License~v2.0~or~later }
                        \tl_set:Nn \l_tmpd_tl { https://www.gnu.org/licenses/old-licenses/gpl-2.0.html }
                      }
                    {lgpl-3.0-or-later}
                      {
                        \tl_set:Nn \l_tmpc_tl { GNU~Lesser~General~Public~License~v3.0~or~later }
                        \tl_set:Nn \l_tmpd_tl { https://www.gnu.org/licenses/lgpl-3.0.html }
                      }
                    {agpl-3.0-or-later}
                      {
                        \tl_set:Nn \l_tmpc_tl { GNU~Affero~General~Public~License~v3.0~or~later }
                        \tl_set:Nn \l_tmpd_tl { https://www.gnu.org/licenses/agpl-3.0.html }
                      }
                  }
                  {
                    \tl_if_empty:NTF \l_tmpb_tl
                      { \tl_set:Nn \l_tmpc_tl { GNU~license } }
                      {
                        \msg_warning:nnxx { zigma } { unknown-license-id } { \tl_use:N \g_zigma_meta_license_id_tl } { gnu }
                        \tl_set:Nx \l_tmpc_tl { GNU~license~(\tl_use:N \g_zigma_meta_license_id_tl) }
                      }
                  }

                \tl_gset:Nx \g_zigma_meta_license_rendered_tl
                  { Released~under~the~\l_tmpc_tl. }

                \tl_if_empty:NT \g_zigma_meta_license_url_tl
                  { \tl_if_empty:NF \l_tmpd_tl { \tl_gset:NV \g_zigma_meta_license_url_tl \l_tmpd_tl } }
              }
            {custom}
              {
                \tl_if_empty:NTF \g_zigma_meta_license_tl
                  {
                    \tl_if_empty:NTF \g_zigma_meta_license_id_tl
                      { \tl_clear:N \g_zigma_meta_license_rendered_tl }
                      { \tl_gset:Nx \g_zigma_meta_license_rendered_tl { License:~\tl_use:N \g_zigma_meta_license_id_tl. } }
                  }
                  { \tl_gset:NV \g_zigma_meta_license_rendered_tl \g_zigma_meta_license_tl }
              }
          }
          {
            \msg_warning:nnx { zigma } { unknown-license-type } { \tl_use:N \g_zigma_meta_license_type_tl }
            \tl_if_empty:NTF \g_zigma_meta_license_tl
              { \tl_clear:N \g_zigma_meta_license_rendered_tl }
              { \tl_gset:NV \g_zigma_meta_license_rendered_tl \g_zigma_meta_license_tl }
          }
      }

    % Optional extra note/url shared by all modes.
    \tl_if_empty:NF \g_zigma_meta_license_note_tl
      {
        \tl_gput_right:Nx \g_zigma_meta_license_rendered_tl
          { ~\tl_use:N \g_zigma_meta_license_note_tl }
      }
    \tl_if_empty:NF \g_zigma_meta_license_url_tl
      {
        \tl_set:NV \l_tmpa_tl \g_zigma_meta_license_rendered_tl
        \tl_gset:Nx \g_zigma_meta_license_rendered_tl
          {
            \exp_not:N \href
