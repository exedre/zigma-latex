%% Zigma Template: Il Cibernetico Journal

%%
%% This template configures Zigma for Il Cibernetico journal format:
%% - Rho plugin (academic journal layout)
%% - Green color theme (#009966)
%% - IEEE citation style
%% - Smart headers (small.title fallback to title + lead author)
%% - Page number in footer center
%%
%% USAGE:
%%   \documentclass[ztemplate=ilcibernetico,zbibfile=refs.bib]{zigma}
%%
%% OVERRIDES:
%%   User options always win. Template sets defaults only.
%%
%% NOTE: This file is loaded from within \ExplSyntaxOn in zigma.cls

\ExplSyntaxOff

%% ============================================================================
%% Class Options (apply only if user didn't specify)
%% ============================================================================

  \def\zigma@plugin{rho}
\else

  \ifx\zigma@plugin\@tempb
    % User chose rho, OK
  \else

%% ============================================================================
%% Post-Setup Hook (apply after \zigmasetup)
%% ============================================================================

\cs_new_protected:Npn \zigma_template_postsetup:
  {
    % Color: Green #009966
    \tl_if_empty:NT \g_zigma_rhocolor_tl
      {
        \tl_gset:Nn \g_zigma_rhocolor_tl { 009966 }
      }
    
    % Box colors: light green background with darker green border
    \tl_if_empty:NT \g_zigma_color_box_bg_tl
      {
        \tl_gset:Nn \g_zigma_color_box_bg_tl { rhocolor!18 }
      }
    \tl_if_empty:NT \g_zigma_color_box_line_tl
      {
        \tl_gset:Nn \g_zigma_color_box_line_tl { rhocolor!60 }
      }
    
    % Text colors: establish hierarchy
    \tl_if_empty:NT \g_zigma_color_text_tl
      {
        \tl_gset:Nn \g_zigma_color_text_tl { black }
      }
    \tl_if_empty:NT \g_zigma_color_author_tl
      {
        \tl_if_empty:NTF \g_zigma_color_text_tl
          { \tl_gset:Nn \g_zigma_color_author_tl { black } }
          { \tl_gset:Nn \g_zigma_color_author_tl { \tl_use:N \g_zigma_color_text_tl } }
      }
    \tl_if_empty:NT \g_zigma_color_section_tl
      {
        \tl_if_empty:NTF \g_zigma_color_text_tl
          { \tl_gset:Nn \g_zigma_color_section_tl { black } }
          { \tl_gset:Nn \g_zigma_color_section_tl { \tl_use:N \g_zigma_color_text_tl } }
      }
    
    % Title colors: establish hierarchy
    \tl_if_empty:NT \g_zigma_color_title_tl
      {
        \tl_if_empty:NTF \g_zigma_color_main_tl
          { \tl_gset:Nn \g_zigma_color_title_tl { green!60!black } }
          { \tl_gset:Nn \g_zigma_color_title_tl { \tl_use:N \g_zigma_color_main_tl } }
      }
    \tl_if_empty:NT \g_zigma_color_subtitle_tl
      {
        \tl_gset:Nn \g_zigma_color_subtitle_tl { \tl_use:N \g_zigma_color_title_tl }
      }
    
    % Link colors: establish hierarchy
    \tl_if_empty:NT \g_zigma_color_url_tl
      {
        \tl_if_empty:NTF \g_zigma_color_accent_tl
          { \tl_gset:Nn \g_zigma_color_url_tl { blue } }
          { \tl_gset:Nn \g_zigma_color_url_tl { \tl_use:N \g_zigma_color_accent_tl } }
      }
    \tl_if_empty:NT \g_zigma_color_journal_tl
      {
        \tl_if_empty:NTF \g_zigma_color_accent_tl
          { \tl_gset:Nn \g_zigma_color_journal_tl { blue } }
          { \tl_gset:Nn \g_zigma_color_journal_tl { \tl_use:N \g_zigma_color_accent_tl } }
      }
    
    % Set link color based on color.url if not already configured
    \tl_if_empty:NT \g_zigma_links_url_color_tl
      {
        \tl_gset:Nx \g_zigma_links_url_color_tl { \tl_use:N \g_zigma_color_url_tl }
      }
    
    % Journal name
    \tl_if_empty:NT \g_zigma_journal_name_tl
      {
        \tl_gset:Nn \g_zigma_journal_name_tl { Il~Cibernetico }
      }
    
    % Header left: small.title with fallback to title
    \tl_if_empty:NT \g_zigma_header_left_tl
      {
        \tl_if_empty:NTF \g_zigma_small_title_tl
          {
            \tl_gset:Nn \g_zigma_header_left_tl 
              { \small\itshape \tl_use:N \g_zigma_title_tl }
          }
          {
            \tl_gset:Nn \g_zigma_header_left_tl 
              { \small\itshape \tl_use:N \g_zigma_small_title_tl }
          }
      }
    
    % Header right: lead author + "et al." if multiple authors
    \tl_if_empty:NT \g_zigma_header_right_tl
      {
        \int_compare:nNnTF { \seq_count:N \g_zigma_author_ids_seq } > { 1 }
          {
            % Multiple authors: lead author et al.
            \tl_gset:Nn \g_zigma_header_right_tl 
              { \small \tl_use:N \g_zigma_leadauthor_tl ~et~al. }
          }
          {
            % Single author: just the name
            \tl_gset:Nn \g_zigma_header_right_tl 
              { \small \tl_use:N \g_zigma_leadauthor_tl }
          }
      }
    
    % Footer center: page number
    \tl_if_empty:NT \g_zigma_footer_center_tl
      {
        \tl_gset:Nn \g_zigma_footer_center_tl { \thepage }
      }
    
    % Corresponding marker: envelope
    \tl_if_empty:NT \g_zigma_corresponding_marker_tl
      {
        \tl_gset:Nn \g_zigma_corresponding_marker_tl { envelope }
      }

  }

\ExplSyntaxOff
